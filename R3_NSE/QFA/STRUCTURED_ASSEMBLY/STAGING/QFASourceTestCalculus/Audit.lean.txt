  nativeTestFunctionOfSource_fderiv φ x

      sourcePartialDeriv φ i x :=
  nativeTestFunctionOfSource_coordinateDeriv φ i x

  sourceTestGradient φ x

      scalarGradient (nativeTestFunctionOfSource φ) (sourceToNative x) :=
  sourceToNative_sourceTestGradient φ x

    (∑ i : Fin 3,
      sourceFieldFunctionOfNative a x i * sourcePartialDeriv φ i x) =
      inner ℝ (a (sourceToNative x))
        (scalarGradient (nativeTestFunctionOfSource φ) (sourceToNative x)) :=
  sourceField_testPairing_eq_nativeInner a φ x
